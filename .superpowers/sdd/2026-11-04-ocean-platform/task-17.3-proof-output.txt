// Qualification output only: descriptor-relative, no-follow and exclusive proof creation.
import path from 'node:path';
import {spawnSync} from 'node:child_process';
const program=String.raw`
import errno,json,os,re,stat,sys
value=json.load(sys.stdin);mode=sys.argv[1];q=value['artifact'];name=value['rehearsal'];override=value.get('proof')
try:
    if not os.path.isabs(q) or os.path.realpath(q)!=q or not re.fullmatch(r'[a-z0-9-]+',name):raise ValueError('invalid_rehearsal_path')
    relative=override if override is not None else name+'/proof.json'
    parts=relative.split('/')
    if len(parts)!=2 or parts[0]!=name or not re.fullmatch(r'[A-Za-z0-9_-][A-Za-z0-9_.-]*\.json',parts[1]):raise ValueError('proof_outside_fresh_rehearsal')
    flags=os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW
    qfd=os.open(q,flags)
    try:
        qstat=os.fstat(qfd)
        if mode=='prepare':os.mkdir(name,0o700,dir_fd=qfd)
        rfd=os.open(name,flags,dir_fd=qfd)
        try:
            rstat=os.fstat(rfd);identity={'qdev':str(qstat.st_dev),'qino':str(qstat.st_ino),'dev':str(rstat.st_dev),'ino':str(rstat.st_ino)}
            if mode=='write':
                if identity!=value['identity']:raise ValueError('proof_parent_changed')
                fd=os.open(parts[1],os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600,dir_fd=rfd)
                try:
                    data=value['body'].encode('utf8');position=0
                    while position<len(data):position+=os.write(fd,data[position:])
                    os.fsync(fd)
                    if os.stat(parts[1],dir_fd=rfd,follow_symlinks=False).st_ino!=os.fstat(fd).st_ino:raise ValueError('proof_identity_changed')
                finally:os.close(fd)
                current=os.stat(name,dir_fd=qfd,follow_symlinks=False)
                if current.st_dev!=rstat.st_dev or current.st_ino!=rstat.st_ino:raise ValueError('proof_parent_changed')
                os.fsync(rfd)
            print(json.dumps({'ok':True,'identity':identity,'relative':relative}))
        finally:os.close(rfd)
    finally:os.close(qfd)
except (ValueError,OSError) as error:
    reason='proof_destination_exists' if isinstance(error,OSError) and error.errno==errno.EEXIST else str(error)
    print(json.dumps({'ok':False,'error':reason}));sys.exit(2)
`;
function invoke(mode,value) {
  const result=spawnSync('python3',['-c',program,mode],{input:JSON.stringify(value),encoding:'utf8',timeout:5000,maxBuffer:4194304});
  let reply;try{reply=JSON.parse(result.stdout);}catch{throw Error('proof_output_helper_failed');}
  if(result.status!==0||!reply.ok)throw Object.assign(Error(reply.error??'proof_output_helper_failed'),{code:reply.error});
  return reply;
}
export function prepareProofOutput(artifact,{rehearsal,proof}={}) {
  const input={artifact: path.resolve(artifact),rehearsal,proof};const prepared=invoke('prepare',input);
  return {root:path.join(input.artifact,rehearsal),proofPath:path.join(input.artifact,prepared.relative),write(value){
    invoke('write',{...input,identity:prepared.identity,body:JSON.stringify(value,null,2)+'\n'});
  }};
}
