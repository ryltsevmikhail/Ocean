import assert from 'node:assert/strict';
import fs from 'node:fs';
import path from 'node:path';
import {builtinModules} from 'node:module';
import {pathToFileURL,fileURLToPath} from 'node:url';
const root=process.cwd();
assert.ok(root.includes('/prefix/lib/ocean/releases/'));
assert.equal(process.env.NODE_PATH,undefined);assert.equal(process.env.NODE_OPTIONS,undefined);
const ts=(await import(process.env.OCEAN_QA_TYPESCRIPT)).default;
const manifest=JSON.parse(fs.readFileSync('manifest.json','utf8'));
const imports=[],unresolved=[],dynamic=[],resourceURLs=[];
for(const name of Object.keys(manifest.files).filter(n=>n.endsWith('.js'))){
 const filename=path.join(root,name),code=fs.readFileSync(filename,'utf8');
 const source=ts.createSourceFile(filename,code,ts.ScriptTarget.Latest,true,ts.ScriptKind.JS);

 const check=(specifier,kind)=>{
  if(specifier.startsWith('node:')||builtinModules.includes(specifier))return;
  try{
   const resolved=fileURLToPath(import.meta.resolve(specifier,pathToFileURL(filename).href));assert.ok(resolved.startsWith(root+'/'),specifier+' escaped installed root');
   assert.ok(fs.existsSync(resolved));imports.push({from:name,specifier,kind,resolved:path.relative(root,resolved)});
  }catch(e){unresolved.push({from:name,specifier,kind,error:e.code??e.message});}
 };
 const visit=node=>{
  if((ts.isImportDeclaration(node)||ts.isExportDeclaration(node))&&node.moduleSpecifier)check(node.moduleSpecifier.text,'static');
  if(ts.isCallExpression(node)&&(node.expression.kind===ts.SyntaxKind.ImportKeyword||(ts.isIdentifier(node.expression)&&node.expression.text==='require'))){
   const arg=node.arguments[0];
   if(arg&&(ts.isStringLiteral(arg)||ts.isNoSubstitutionTemplateLiteral(arg)))check(arg.text,'dynamic-literal');
   else dynamic.push({from:name,expression:node.getText(source).slice(0,250)});
  }
  if(ts.isNewExpression(node)&&ts.isIdentifier(node.expression)&&node.expression.text==='URL'&&node.arguments?.[1]?.getText(source)==='import.meta.url'){
   const arg=node.arguments[0];if(arg&&(ts.isStringLiteral(arg)||ts.isNoSubstitutionTemplateLiteral(arg))){if(arg.text.startsWith('.'))check(arg.text,'module-relative-URL');}else resourceURLs.push({from:name,expression:node.getText(source).slice(0,250)});
  }
  ts.forEachChild(node,visit);
 };visit(source);
}

 providers[name]={url,exports:Object.keys(module).sort()};
}
